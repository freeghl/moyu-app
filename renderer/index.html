45import pandas as pd
import numpy as np
from datetime import datetime
from typing import List, Dict, Optional

class DataProcessor:
    """处理和分析数据的核心模块"""

    def __init__(self, config: Dict):
        self.config = config
        self.logger = logging.getLogger(__name__)
        self._cache = {}

    def process_batch(self, data: List[Dict]) -> pd.DataFrame:
        """批量处理数据记录"""
        self.logger.info(f"Processing batch of {len(data)} records")

        df = pd.DataFrame(data)
        df['timestamp'] = pd.to_datetime(df['timestamp'])
        df['value'] = df['value'].apply(self._normalize)

        # 过滤异常值
        mask = np.abs(df['value'] - df['value'].mean()) < 3 * df['value'].std()
        df_clean = df[mask].copy()

        self.logger.info(f"Cleaned: {len(df) - len(df_clean)} outliers removed")
        return df_clean

    def _normalize(self, value: float) -> float:
        """Min-max 归一化"""
        min_val = self.config.get('min_value', 0)
        max_val = self.config.get('max_value', 1)
        return (value - min_val) / (max_val - min_val + 1e-8)

    def generate_report(self, df: pd.DataFrame) -> Dict:
        """生成数据分析报告"""
        report = {
            'total_records': len(df),
            'mean_value': df['value'].mean(),
            'std_value': df['value'].std(),
            'generated_at': datetime.now().isoformat(),
        }
        return report
